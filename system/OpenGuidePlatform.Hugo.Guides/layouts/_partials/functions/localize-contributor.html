{{- /* Purpose: Display a contributor record in one language. Input: dict "record" (contributor map) and "language". Returns: the record, with name replaced by localizedNames. when present. */ -}} {{- $record := .record -}} {{- $language := lower (string .language) -}} {{- $name := "" -}} {{- with $record.localizedNames -}} {{- range $key, $value := . -}} {{- if and (eq (lower $key) $language) $value -}}{{- $name = $value -}}{{- end -}} {{- end -}} {{- end -}} {{- if $name -}} {{- $record = merge $record (dict "name" $name) -}} {{- end -}} {{- return $record -}}
